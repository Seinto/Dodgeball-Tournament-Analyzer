Googleスプレッドシート連携の手順 & GASコード
  1. Googleドライブで新しい「Googleスプレッドシート」を作成します。
  2. メニューの [拡張機能] > [Apps Script] を開きます。
  3. 表示されたエディタの内容を全て消去し、以下のGASコードを貼り付けて保存します。
  4. 右上の [デプロイ] > [新しいデプロイ] を選択します。
  5. 種類で「ウェブアプリ」を選択し、アクセスできるユーザーを「全員」に設定してデプロイします。
  6. 発行された Web App URL を本アプリの設定画面に入力してください。
GASコード（読み書き対応最新版） 
function doPost(e) {
  try {
    if (!e || !e.postData || !e.postData.contents) {
      throw new Error("送信されたデータが空です。");
    }

    var contents = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();

    var tournaments = contents.tournaments;
    if (!tournaments) {
      tournaments = [{
        index: 1,
        label: contents.label || "試合データ",
        matches: contents.matches || [],
        players: contents.players || [],
        previousSheetName: null
      }];
    }

    tournaments.forEach(function(tour) {
      writeTournamentSheet(ss, tour);
    });

    return ContentService.createTextOutput(JSON.stringify({ status: "success" }))
      .setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function writeTournamentSheet(ss, tour) {
  var label = tour.label || "試合データ";
  var sheetName = sanitizeSheetName(label);
  var matches = tour.matches || [];
  var players = tour.players || [];

  if (tour.previousSheetName && tour.previousSheetName !== sheetName) {
    var oldSheet = ss.getSheetByName(tour.previousSheetName);
    if (oldSheet) ss.deleteSheet(oldSheet);
  }

  var sheet = ss.getSheetByName(sheetName);
  if (!sheet) {
    sheet = ss.insertSheet(sheetName);
  } else {
    sheet.clearContents();
  }

  var rows = [];
  matches.forEach(function(m, idx) {
    var oppName = m.opp || ("対戦相手" + (idx + 1));
    var resultText = m.result || ((m.us || 0) > (m.them || 0) ? "勝" : ((m.us || 0) < (m.them || 0) ? "負" : "引き分け"));

    rows.push(["【第 " + (idx + 1) + " 試合スタッツ】", "", "", "", "", "", "", ""]);
    rows.push(["対戦相手", oppName, "", "", "", "", "", ""]);
    rows.push(["結果（見方-相手）", (m.us || 0) + "-" + (m.them || 0) + " (" + resultText + ")", "", "", "", "", "", ""]);
    rows.push(["ボールキープ時間", keepStrToSeconds(m.keep) + "秒", "", "", "", "", "", ""]);
    rows.push(["備考", m.notes || "", "", "", "", "", "", ""]);
    rows.push(["", "", "", "", "", "", "", ""]);
    rows.push(["", "", "DF", "DF", "内野AT", "内野AT", "外野AT", "外野AT"]);
    rows.push(["背番号", "選手名", "○", "×", "○", "×", "○", "×"]);

    players.forEach(function(p, pIdx) {
      var stat = (p.stats && p.stats[idx]) ? p.stats[idx] : null;
      var def = (stat && stat.def) || [0, 0];
      var ia = (stat && stat.ia) || [0, 0];
      var oa = (stat && stat.oa) || [0, 0];
      rows.push([pIdx + 1, p.name, def[0] || 0, def[1] || 0, ia[0] || 0, ia[1] || 0, oa[0] || 0, oa[1] || 0]);
    });

    if (idx < matches.length - 1) {
      rows.push(["", "", "", "", "", "", "", ""]);
      rows.push(["", "", "", "", "", "", "", ""]);
    }
  });

  if (rows.length > 0) {
    sheet.getRange(1, 1, rows.length, 8).setValues(rows);
  }
}

function sanitizeSheetName(label) {
  var name = String(label || "試合データ").replace(/[\[\]\*\/\\\?:]/g, "_");
  if (name.length > 90) name = name.substring(0, 90);
  return name || "試合データ";
}

function keepStrToSeconds(keep) {
  if (!keep) return 0;
  var parts = String(keep).split(":");
  if (parts.length === 2) {
    return (parseInt(parts[0], 10) || 0) * 60 + (parseInt(parts[1], 10) || 0);
  }
  if (parts.length === 3) {
    return (parseInt(parts[0], 10) || 0) * 3600 + (parseInt(parts[1], 10) || 0) * 60 + (parseInt(parts[2], 10) || 0);
  }
  return parseInt(keep, 10) || 0;
}

function doGet(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheets = ss.getSheets();
    var tournaments = {};

    sheets.forEach(function(sheet) {
      var label = sheet.getName();
      var data = sheet.getDataRange().getValues();
      if (!data || data.length < 8) return;

      var matches = [];
      var playersMap = {}; 
      var currentMatchIdx = -1;

      for (var i = 0; i < data.length; i++) {
        var row = data[i];
        if (row[0] && String(row[0]).indexOf("【第") === 0) {
          currentMatchIdx++;
          var opp = "", us = 0, them = 0, result = "勝", keep = "0:00", notes = "";
          
          if (i + 1 < data.length) opp = data[i + 1][1] || "";
          if (i + 2 < data.length) {
            var resStr = String(data[i + 2][1] || "");
            var mRes = resStr.match(/(\d+)-(\d+)\s*\((.*?)\)/);
            if (mRes) {
              us = parseInt(mRes[1], 10) || 0;
              them = parseInt(mRes[2], 10) || 0;
              result = mRes[3] || "勝";
            }
          }
          if (i + 3 < data.length) {
            var kSec = parseInt(String(data[i + 3][1] || "").replace("秒", ""), 10) || 0;
            var mSec = Math.floor(kSec / 60);
            var sSec = kSec % 60;
            keep = mSec + ":" + (sSec < 10 ? "0" : "") + sSec;
          }
          if (i + 4 < data.length) notes = data[i + 4][1] || "";

          matches.push({ opp: opp, us: us, them: them, keep: keep, result: result, notes: notes });
        }

        if (row[0] === "背番号" && row[1] === "選手名") {
          for (var j = i + 1; j < data.length; j++) {
            var pRow = data[j];
            if (!pRow[1] || String(pRow[0]).indexOf("【第") === 0) break; 
            var pName = String(pRow[1]).trim();
            if (!pName) continue;

            if (!playersMap[pName]) {
              playersMap[pName] = { name: pName, stats: [] };
            }

            var dO = parseInt(pRow[2], 10) || 0;
            var dX = parseInt(pRow[3], 10) || 0;
            var iO = parseInt(pRow[4], 10) || 0;
            var iX = parseInt(pRow[5], 10) || 0;
            var oO = parseInt(pRow[6], 10) || 0;
            var oX = parseInt(pRow[7], 10) || 0;

            playersMap[pName].stats[currentMatchIdx] = {
              def: [dO, dX],
              ia: [iO, iX],
              oa: [oO, oX]
            };
          }
        }
      }

      if (matches.length > 0) {
        var players = Object.values(playersMap);
        tournaments[label] = {
          label: label,
          matches: matches,
          players: players,
          syncedSheetName: label
        };
      }
    });

    return ContentService.createTextOutput(JSON.stringify({ status: "success", data: tournaments }))
      .setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
Gemini APIキーの取得方法
  1. Google AI Studio にアクセス
    https://aistudio.google.com/app/apikey へ移動します。
  2. Googleアカウントでログイン
    お持ちのGoogleアカウントでサインインします。
  3. APIキーを作成
    「Create API key」ボタンを押します。
  4. キーをコピー＆ペースト
    発行された文字列（`AIzaSy...`）をコピーし、設定画面の入力欄へ貼り付けて保存してください。

※基本的な利用枠であれば無料で作成できます。
